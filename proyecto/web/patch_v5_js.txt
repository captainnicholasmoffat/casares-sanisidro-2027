  function firstSentence(t, max){

    if (s.length > (max || 170)) s = s.slice(0, (max || 170) - 5).replace(/\s+\S*$/, "") + "…";

  function markedP(tag, text, marks, cls){
    var p = el(tag, cls ? {"class": cls} : null);
    if (!marks || !marks.length) { p.textContent = text; return p; }
    var ms = marks.slice().sort(function(a, b){ return a[0] - b[0] || b[1] - a[1]; }), pos = 0;
    ms.forEach(function(m){
      if (m[0] < pos || m[1] > text.length) return;
      if (m[0] > pos) p.appendChild(document.createTextNode(text.slice(pos, m[0])));
      p.appendChild(el("span", {"class": m[2] === "b" ? "mb" : (m[2] === "sg" ? "ms" : "mi")}, text.slice(m[0], m[1])));
      pos = m[1];
    });
    if (pos < text.length) p.appendChild(document.createTextNode(text.slice(pos)));
    return p;
  }
